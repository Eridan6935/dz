import random

def is_valid_pin(pin: str) -> bool:
    """
    Проверяет ПИН-код на безопасность:
    1. Не должно быть ситуаций, когда все цифры одинаковые (например, 1111).
    2. Не должно быть последовательностей строго по порядку (например, 1234).
    """
    # 1. Проверка на одинаковые цифры
    # Если символ, повторенный 4 раза, равен всей строке — код из одинаковых цифр
    if pin[0] * 4 == pin:
        return False
    
    # 2. Проверка на последовательный порядок цифр
    # Превращаем символы в числа для проверки условий ровно по порядку
    d0 = int(pin[0])
    d1 = int(pin[1])
    d2 = int(pin[2])
    d3 = int(pin[3])
    
    if (d1 == d0 + 1) and (d2 == d1 + 1) and (d3 == d2 + 1):
        return False
        
    return True

def generate_pin() -> str:
    """Генерирует случайный валидный 4-значный ПИН-код."""
    while True:
        # Генерируем случайный ПИН-код из 4 цифр
        raw_pin = f"{random.randint(0, 9999):04d}"
        
        # Проверяем на соответствие правилам безопасности
        if is_valid_pin(raw_pin):
            return raw_pin

# Пример использования:
if __name__ == "__main__":
    pin = generate_pin()
    print(f"Сгенерированный безопасный ПИН-код: {pin}")
